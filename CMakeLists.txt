if(NGX_COMMON_INCLUDE AND NGX_NR_INCLUDE)
    find_program(CLANG_CL clang-cl REQUIRED)
    set(NGX_PARAMETER_OBJECT "${CMAKE_BINARY_DIR}/NgxParameterAccess.obj")
    add_custom_command(OUTPUT "${NGX_PARAMETER_OBJECT}"
        COMMAND "${CLANG_CL}" /nologo /std:c++17 /O2 /c /GR- /EHs-c- /Zl
            "/I${NGX_COMMON_INCLUDE}" "/I${NGX_NR_INCLUDE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tools/NgxParameterAccess.cpp" "/Fo${NGX_PARAMETER_OBJECT}"
        DEPENDS tools/NgxParameterAccess.cpp VERBATIM)
    set_source_files_properties("${NGX_PARAMETER_OBJECT}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
    add_library(neural_backend STATIC src/backend/NeuralBackend.cpp src/backend/GpuInterop.cpp src/backend/NgxLifetime.cpp "${NGX_PARAMETER_OBJECT}")
    target_include_directories(neural_backend PRIVATE "${NGX_COMMON_INCLUDE}" "${NGX_NR_INCLUDE}")
    target_compile_definitions(neural_backend PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    target_compile_options(neural_backend PRIVATE -Wall -Wextra -Werror -Wno-cast-function-type -Wno-missing-field-initializers)
    target_link_libraries(neural_backend PUBLIC d3d11 d3d12 d3dcompiler dxgi dxguid)
    target_link_libraries(neural_layer PRIVATE neural_backend)
    target_compile_definitions(neural_layer PRIVATE TF2VRNR_BACKEND=1)
    add_executable(replay_probe tools/ReplayProbe.cpp src/layer/Support.cpp)
    set_target_properties(replay_probe PROPERTIES OUTPUT_NAME ReplayProbe)
    target_compile_definitions(replay_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    target_link_libraries(replay_probe PRIVATE neural_backend)
    target_link_options(replay_probe PRIVATE -municode -static -static-libgcc -static-libstdc++)
    set(NGX_FG_INCLUDE "" CACHE PATH "External NVIDIA Frame Generation SDK include directory")
    if(NGX_FG_INCLUDE)
        target_sources(neural_backend PRIVATE src/backend/FrameGeneration.cpp)
        target_include_directories(neural_backend PRIVATE "${NGX_FG_INCLUDE}")
        target_compile_definitions(neural_layer PRIVATE TF2VRNR_FRAMEGEN=1)
        target_sources(neural_layer PRIVATE src/layer/FrameDelivery.cpp)
        add_executable(frame_generation_probe tools/FrameGenerationProbe.cpp src/layer/Support.cpp)
        set_target_properties(frame_generation_probe PROPERTIES OUTPUT_NAME FrameGenerationProbe)
        target_include_directories(frame_generation_probe PRIVATE "${NGX_COMMON_INCLUDE}" "${NGX_NR_INCLUDE}" "${NGX_FG_INCLUDE}")
        target_compile_definitions(frame_generation_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
        target_link_libraries(frame_generation_probe PRIVATE neural_backend)
        target_compile_options(frame_generation_probe PRIVATE -Wall -Wextra -Werror -Wno-cast-function-type -Wno-missing-field-initializers)
        target_link_options(frame_generation_probe PRIVATE -municode -static -static-libgcc -static-libstdc++)
        add_executable(frame_generation_backend_probe tools/FrameGenerationBackendProbe.cpp src/layer/Support.cpp)
        set_target_properties(frame_generation_backend_probe PROPERTIES OUTPUT_NAME FrameGenerationBackendProbe)
        target_compile_definitions(frame_generation_backend_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
        target_link_libraries(frame_generation_backend_probe PRIVATE neural_backend)
        target_link_options(frame_generation_backend_probe PRIVATE -municode -static -static-libgcc -static-libstdc++)
        add_executable(frame_generation_openxr_probe tools/FrameGenerationOpenXRProbe.cpp src/layer/FrameDelivery.cpp src/layer/Support.cpp)
        set_target_properties(frame_generation_openxr_probe PROPERTIES OUTPUT_NAME FrameGenerationOpenXRProbe)
        target_include_directories(frame_generation_openxr_probe PRIVATE "${OPENXR_INCLUDE_DIR}")
        target_compile_definitions(frame_generation_openxr_probe PRIVATE XR_NO_PROTOTYPES WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
        target_link_libraries(frame_generation_openxr_probe PRIVATE neural_backend)
        target_link_options(frame_generation_openxr_probe PRIVATE -municode -static -static-libgcc -static-libstdc++)
        add_executable(frame_generation_capture_probe tools/FrameGenerationCaptureProbe.cpp src/layer/DepthCapture.cpp src/layer/Graphics.cpp src/layer/Bindings.cpp src/layer/Support.cpp src/layer/CaptureWriter.cpp)
        set_target_properties(frame_generation_capture_probe PROPERTIES OUTPUT_NAME FrameGenerationCaptureProbe)
        target_compile_definitions(frame_generation_capture_probe PRIVATE TF2VRNR_BACKEND=1 TF2VRNR_FRAMEGEN=1 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)

    target_include_directories(${target} PRIVATE "${OPENXR_INCLUDE_DIR}")
    target_compile_definitions(${target} PRIVATE XR_NO_PROTOTYPES WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-cast-function-type -Wno-missing-field-initializers)
        target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()

add_custom_target(dist_package
    COMMAND ${CMAKE_COMMAND} -E make_directory "${PACKAGE_DIR}/plugins/layer" "${PACKAGE_DIR}/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/package/mods" "${PACKAGE_DIR}/mods"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:neural_plugin>" "${PACKAGE_DIR}/plugins/"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:neural_setup>" "${PACKAGE_DIR}/"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:neural_layer>" "${CMAKE_CURRENT_SOURCE_DIR}/config/TF2VRNeuralRendering.ini"
        "${CMAKE_CURRENT_SOURCE_DIR}/manifest/XR_APILAYER_TF2VR_neural_rendering.json" "${PACKAGE_DIR}/plugins/layer/"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/README.md" "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE"
        "${CMAKE_CURRENT_SOURCE_DIR}/THIRD-PARTY-NOTICES.md" "${CMAKE_CURRENT_SOURCE_DIR}/tools/capture.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_runtime.py" "${CMAKE_CURRENT_SOURCE_DIR}/tools/control.py" "${PACKAGE_DIR}/"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/docs" "${PACKAGE_DIR}/docs"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "${PACKAGE_DIR}/licenses"
    COMMAND ${CMAKE_COMMAND} -E chdir "${CMAKE_BINARY_DIR}/package" ${CMAKE_COMMAND} -E tar cf
        "${CMAKE_BINARY_DIR}/${PACKAGE_NAME}.zip" --format=zip "${PACKAGE_NAME}"
    DEPENDS neural_layer neural_plugin neural_setup VERBATIM)
